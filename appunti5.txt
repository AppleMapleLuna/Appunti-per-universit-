class Report:
    def __init__(self,nome,file):
        self.nome = nome 
        self.file = file 

    def __getstate__(self):     # Dice a pickle i campi dell'oggetto che vogliamo considerare
        return {"nome":self.nome}

    def __setstate__(self,dati):
        self.nome = dati("nome")
        self.file = None
